# Fix npm EACCES (node_modules owned by root) - run WITHOUT sudo

# 1. Diagnose ownership
ls -ld node_modules
ls -ld ~/.npm

# 2. Fix ownership (replace path if different)
sudo chown -R $(whoami):$(whoami) /home/rash/Code/PROJECTS/PERSONAL-WEBSITE
sudo chown -R $(whoami):$(whoami) ~/.npm

# 3. Clean reinstall (no sudo)
rm -rf node_modules package-lock.json
npm install

# 4. Verify
npm run build
npm run dev
# -> http://localhost:5173 (dev) / http://localhost:4173 (preview)
